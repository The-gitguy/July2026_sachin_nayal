Kubernetes — Deploy Full Application Stack


apiVersion: apps/v1
kind: Deployment
metadata:
  name: sachin-app
spec:
  replicas: 2
  selector:
    matchLabels:
      app: sachin-app
  template:
    metadata:
      labels:
        app: sachin-app
    spec:
      containers:
        - name: nginx
          image: nginx
          ports:
            - containerPort: 80
---
apiVersion: v1
kind: Service
metadata:
  name: sachin-service
spec:
  selector:
    app: sachin-app
  ports:
    - port: 80
      targetPort: 80
