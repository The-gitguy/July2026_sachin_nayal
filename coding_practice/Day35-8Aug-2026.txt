# 🔹 Jenkins Pipeline

```groovy
pipeline {
    agent any

    stages {
        stage('Checkout') {
            steps {
                git branch: 'main',
                    url: 'https://github.com/sachin/sample-java-app.git'
            }
        }

        stage('Build') {
            steps {
                sh '''
                    mvn clean compile || {
                        echo "Compile failed"
                        exit 1
                    }
                '''
            }
        }

        stage('Test') {
            steps {
                sh '''
                    mvn test || {
                        echo "Tests failed"
                        exit 1
                    }
                '''
            }
        }
    }
}
```
