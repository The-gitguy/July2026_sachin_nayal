# Day17 - Loops, ALB , Namespace, Top Processes

## Ansible - Create Multiple users in Loop

---
- name: Create multiple Users
  hosts: all
  become: yes

  vars:
    users:
      -dev1
      -dev2
      -dev3

  
  tasks:
    - name: Create users
      user:
        name:"{{ item }}"
        state: present
      loop: "{{ users }}"

  name: sachin-deploy
  namespace: sachin-namespace

  replica: 2
  selector:
    matchlabels:
      app: sachin

    metadata:
      labels:
        app: sachin

# Deployment Simulation

## Jenkins Pipeline -Simulated deployment
"Simulate Deployment to different environments "

pipeline{
  agent any
  parameters{
    choice(name: 'ENVIRONMENT' , choices:['dev','staging','prod'], description : 'Deployment Environment')
  }
  environment {
        INSTITUTE_NAME= "sachin"

  }
  stages{
    stage('Deploy'){
        steps {
            sh 'echo "Deployment $INSTITUTE_NAME application to $ENVIRONMENT"'
        }
    }
  }
}
