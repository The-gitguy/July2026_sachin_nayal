# JENKINS PIPELINE — USING CREDENTIALS

pipeline {
    agent any

    environment {
        INSTITUTE_NAME = "sachin"
    }

    stages {
        stage('Checkout') {
            steps {
                git url: 'https://github.com/sachin/sample-java-app.git'
            }
        }

        stage('Use Secret') {
            steps {
                withCredentials([
                    string(
                        credentialsId: 'sachin_API_KEY',
                        variable: 'API_KEY'
                    )
                ]) {
                    sh 'echo "Using API key for $INSTITUTE_NAME"'
                    sh './run-app.sh'
                }
            }
        }
    }
}
