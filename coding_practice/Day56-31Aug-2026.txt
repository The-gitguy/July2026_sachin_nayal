## Rollback Strategies
# Jenkins Pipeline

pipeline{
    agent any

    environment {
        INSTITUTE_NAME = "Sachin" 
        TEAM = "DevOps" 
        ENV = "prod"
    }

    stages {
        stage('Deploy'){
            script{
                try { sh 'echo "Deploying $INSTITUTE_NAME app in $ENV for TEAM"'  
                      sh 'exit 1' // simulate failure 
                    } 
                    catch (Exception e) {
                       echo "Deployment failed. Rolling back $INSTITUTE_NAME app in $ENV" }
            }
        }
    }
}
